2Methods Text Generator

40def _resolve_analysis_citation(name: str) -> str: 

41 """Return canonical in-text citation string for an analysis name, or empty string.""" 

42 norm = name.strip().lower() 

43 for keys, citation in _ANALYSIS_CITATIONS: 

44 if any(k in norm for k in keys): 

45 return citation 

46 return ""

49class MethodsGenerator: 

50 """Generates methods text for publication.""" 

51 

52 def generate_methods_paragraph( 

53 self, 

54 analyses_performed: List[Dict[str, Any]], 

55 file_format: str = "", 

56 n_cells: int = 0, 

57 n_recordings: int = 0, 

58 ) -> str: 

59 """ 

60 Generate a complete methods paragraph. 

61 

62 Parameters 

63 ---------- 

64 analyses_performed : list of dict 

65 Each dict has keys: 'name', 'parameters' (dict of param:value) 

66 file_format : str 

67 Recording file format (e.g., "Axon Binary Format (ABF)") 

68 n_cells : int 

69 Number of cells analyzed 

70 n_recordings : int 

71 Number of recordings 

72 """ 

73 version = getattr(synaptipy, "__version__", "unknown") 

74 

75 parts = [] 

76 parts.append( 

77 f"Electrophysiology data were analyzed using Synaptipy v{version} " 

78 f"(Shahul, 2026; https://github.com/anzalks/synaptipy) " 

79 f"running on Python {platform.python_version()} " 

80 f"with Neo I/O (Garcia et al., 2014) and SciPy/NumPy (Virtanen et al., 2020; Harris et al., 2020). " 

81 ) 

82 

83 if file_format: 

84 parts.append(f"Recordings were acquired in {file_format} format. ") 

85 

86 if n_cells or n_recordings: 

87 parts.append(f"A total of {n_recordings} recordings from {n_cells} cells were analyzed. ") 

88 

89 for analysis in analyses_performed: 

90 name = analysis.get("name", "") 

91 params = analysis.get("parameters", {}) 

92 if name and params: 

93 param_str = ", ".join(f"{k}={v}" for k, v in sorted(params.items())) 

94 cite = _resolve_analysis_citation(name) 

95 cite_suffix = f" ({cite})" if cite else "" 

96 parts.append(f"{name} was performed with parameters: {param_str}{cite_suffix}. ") 

97 

98 parts.append( 

99 f"Analysis was performed on {datetime.now().strftime('%Y-%m-%d')}. " 

100 f"Full parameter sets and raw results are available in the " 

101 f"exported NWB files (Rubel et al., 2022)." 

102 ) 

103 

104 return "".join(parts)

115 def generate_analysis_summary( 

116 self, 

117 analyses_performed: List[Dict[str, Any]], 

118 ) -> str: 

119 """Generate a structured summary table (markdown) of all analyses.""" 

120 version = getattr(synaptipy, "__version__", "unknown") 

121 lines = [ 

122 f"| Software | Synaptipy v{version} |", 

123 f"| Python | {platform.python_version()} |", 

124 f"| Platform | {platform.system()} {platform.release()} |", 

125 f"| Date | {datetime.now().strftime('%Y-%m-%d %H:%M')} |", 

126 "", 

127 "| Analysis | Parameters |", 

128 "|----------|-----------|", 

129 ] 

130 for a in analyses_performed: 

131 name = a.get("name", "unknown") 

132 params = a.get("parameters", {}) 

133 param_str = "; ".join(f"{k}={v}" for k, v in params.items()) 

134 lines.append(f"| {name} | {param_str} |") 

135 

136 return "\n".join(lines)
